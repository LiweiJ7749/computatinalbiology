# ============================================================
# SpaGCN + SpaSEG 环境依赖（已合并 & 解决冲突）
# ------------------------------------------------------------
# 目标 Python 版本：3.9（SpaSEG 强制要求 Python==3.9；SpaGCN 兼容）
# 用途：在本地对比 SpaGCN 与 SpaSEG 两个深度学习模型的效果
#
# 冲突解决说明：
#   1. torch  -> 统一升级为 CUDA 版 torch 2.5.1+cu124，支持本机 RTX4060
#               (Ada, sm_89)。注意：+cu124 构建不在普通 PyPI，PyPI/清华镜像
#               默认给 CPU 版（torch==2.5.1+cpu）。必须从 PyTorch CUDA wheel
#               镜像安装：
#                 python -m pip install "torch==2.5.1+cu124" \
#                     -f https://mirrors.aliyun.com/pytorch-wheels/cu124/
#   2. scanpy -> SpaGCN 需 >1.5；SpaSEG 需 >=1.9.1 → 取 >=1.9.1
#   3. numpy / numba -> squidpy 1.2.3 要求 numba>=0.55,<0.56 → 用 0.55.2；
#               numba 0.55 需 numpy<1.23，故锁定 numpy==1.22.4
#               （torch 2.5.1+cu124 的 numpy 约束与 1.22.4 兼容，未升级 numpy）
#   4. louvain  -> SpaGCN 1.2.7 要求 louvain>=0.8.0；0.8.1 依赖新版
#               igraph>=0.10,<0.11（替代旧 python-igraph），故不再锁旧包名
#   5. cell2location -> 仅 SpaSEG 下游"细胞反卷积"分析需要，且依赖 jax
#               （Windows 兼容差，是已知环境冲突源），默认注释，按需启用
#   6. GPU 化：SpaGCN 使用本地补丁源码 src/vendor/SpaGCN_src（run 脚本
#              通过 sys.path 优先加载），SpaSEG 原生支持 device 参数。
# ============================================================

# ---- 深度学习框架（CUDA 版 torch，支持 RTX4060/Ada sm_89）----
torch==2.5.1+cu124

# ---- 数值 / 科学计算 ----
# numpy 1.22.4：满足 numba 0.55（需 <1.23）与 SpaSEG 的 >=1.22.4
numpy==1.22.4
pandas>=1.5.1
scipy>=1.7.3
scikit-learn>=1.2.2

# ---- 单细胞 / 空间转录组生态 ----
scanpy>=1.9.1
anndata>=0.8.0
matplotlib>=3.6.0
seaborn>=0.11.0
statsmodels==0.14.0

# ---- SpaGCN 专属依赖 ----
# SpaGCN 1.2.7 setup.py 要求 louvain>=0.8.0；louvain 0.8.1 依赖新版
# igraph>=0.10,<0.11（PyPI 新包名 "igraph"，已取代旧 "python-igraph"），
# 两者提供同一个 igraph 模块，故不再单独锁定旧版 python-igraph
louvain==0.8.1
natsort>=7.1.1

# ---- SpaSEG 专属依赖 ----
# README 写的 4.7.0 在 PyPI 实为 4.7.0.72（4 段式版本号）
opencv-python==4.7.0.72
squidpy==1.2.3
shapely==2.0.1
# numba 0.55.2：满足 squidpy 1.2.3（需 >=0.55,<0.56）与 SpaSEG 的 >=0.55.2，
# 且与 numpy 1.22.4 兼容
numba==0.55.2

# ---- 可选（SpaSEG 下游细胞反卷积才需要；依赖 jax，Windows 易冲突）----
# cell2location==0.8a0
